cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.5" CACHE STRING "macOS development application target" FORCE)
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" SOFTLOAF_VERSION LIMIT_COUNT 1)
if(NOT SOFTLOAF_VERSION MATCHES "^([0-9]+\\.[0-9]+\\.[0-9]+)(-[A-Za-z0-9.-]+)?$")
    message(FATAL_ERROR "Invalid VERSION")
endif()
set(SOFTLOAF_BASE_VERSION "${CMAKE_MATCH_1}")
project(SoftLoafSpectral VERSION ${SOFTLOAF_BASE_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8 /Zc:__cplusplus)
endif()

add_library(softloaf_core SHARED core/src/core.cpp)
target_include_directories(softloaf_core PUBLIC core/include)
target_compile_features(softloaf_core PUBLIC cxx_std_23)
set_target_properties(softloaf_core PROPERTIES CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(softloaf_core PRIVATE /W4 /fp:precise)
else()
    target_compile_options(softloaf_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math)
endif()

find_package(PNG REQUIRED)
add_library(softloaf_imageio SHARED core/imageio/png_stream.cpp)
target_compile_features(softloaf_imageio PRIVATE cxx_std_23)
target_link_libraries(softloaf_imageio PRIVATE PNG::PNG)
set_target_properties(softloaf_imageio PROPERTIES CXX_VISIBILITY_PRESET hidden)

include(cmake/RawTherapee.cmake)
include(CTest)
if(WIN32)
    add_library(softloaf_d3d12 SHARED core/d3d12/d3d12_backend.cpp)
    target_compile_features(softloaf_d3d12 PRIVATE cxx_std_23)
    target_link_libraries(softloaf_d3d12 PRIVATE d3d12 dxgi d3dcompiler)
    target_compile_definitions(softloaf_d3d12 PRIVATE UNICODE _UNICODE)
endif()
if(APPLE)
    enable_language(OBJCXX)
    add_library(softloaf_metal SHARED core/metal/metal_backend.mm)
    target_compile_features(softloaf_metal PRIVATE cxx_std_23)
    target_compile_options(softloaf_metal PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(softloaf_metal PRIVATE "-framework Foundation" "-framework Metal")
    set_target_properties(softloaf_metal PROPERTIES CXX_VISIBILITY_PRESET hidden)
endif()
if(BUILD_TESTING)
    add_executable(softloaf_core_test core/tests/core_test.cpp)
    target_link_libraries(softloaf_core_test PRIVATE softloaf_core)
    target_compile_features(softloaf_core_test PRIVATE cxx_std_23)
    add_test(NAME core_hand_calculated COMMAND softloaf_core_test)
    add_executable(softloaf_process_tree_test core/tests/process_tree_test.cpp)
    target_compile_features(softloaf_process_tree_test PRIVATE cxx_std_23)
    add_test(NAME desktop_service_ownership COMMAND softloaf_process_tree_test)
    if(WIN32)
        add_executable(softloaf_windows_dpi_test WIN32 core/tests/windows_dpi_test.cpp)
        target_sources(softloaf_windows_dpi_test PRIVATE desktop/windows/app.manifest)
        target_link_libraries(softloaf_windows_dpi_test PRIVATE user32)
        add_test(NAME desktop_per_monitor_dpi COMMAND softloaf_windows_dpi_test)
    endif()
endif()

option(SOFTLOAF_DESKTOP "Build the local development application" ON)
if(WIN32 AND SOFTLOAF_DESKTOP)
    enable_language(RC)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(SOFTLOAF_WEBVIEW2_SDK "" CACHE PATH "Extracted official Microsoft.Web.WebView2 SDK")
    set(SOFTLOAF_WEBVIEW2_CACHE "" CACHE PATH "WebView2 profile on the configured external pixel-data volume")
    find_path(SOFTLOAF_WEBVIEW2_INCLUDE WebView2.h HINTS "${SOFTLOAF_WEBVIEW2_SDK}/build/native/include" REQUIRED)
    find_library(SOFTLOAF_WEBVIEW2_LOADER WebView2LoaderStatic HINTS "${SOFTLOAF_WEBVIEW2_SDK}/build/native/x64" REQUIRED)
    add_executable(SoftLoafSpectralWindows WIN32 desktop/windows/main.cpp desktop/windows/app.manifest)
    configure_file(desktop/windows/application.rc.in windows-application.rc @ONLY)
    target_sources(SoftLoafSpectralWindows PRIVATE "${CMAKE_BINARY_DIR}/windows-application.rc")
    target_compile_features(SoftLoafSpectralWindows PRIVATE cxx_std_23)
    target_include_directories(SoftLoafSpectralWindows PRIVATE "${SOFTLOAF_WEBVIEW2_INCLUDE}")
    target_link_libraries(SoftLoafSpectralWindows PRIVATE "${SOFTLOAF_WEBVIEW2_LOADER}" user32 shell32 ole32 oleaut32 shlwapi version)
    target_compile_definitions(SoftLoafSpectralWindows PRIVATE UNICODE _UNICODE)
    set_target_properties(SoftLoafSpectralWindows PROPERTIES OUTPUT_NAME "SoftLoaf Spectral")
    add_dependencies(SoftLoafSpectralWindows softloaf_core softloaf_imageio softloaf_d3d12)
    if(BUILD_TESTING)
        add_executable(softloaf_windows_branding_test core/tests/windows_branding_test.cpp)
        target_compile_definitions(softloaf_windows_branding_test PRIVATE UNICODE _UNICODE)
        target_link_libraries(softloaf_windows_branding_test PRIVATE user32 version)
        add_dependencies(softloaf_windows_branding_test SoftLoafSpectralWindows)
        add_test(NAME desktop_embedded_branding COMMAND softloaf_windows_branding_test "$<TARGET_FILE:SoftLoafSpectralWindows>")
    endif()
    file(GENERATE OUTPUT "$<TARGET_FILE_DIR:SoftLoafSpectralWindows>/runtime.ini" CONTENT
        "[runtime]\nsource=${CMAKE_SOURCE_DIR}\npython=${Python3_EXECUTABLE}\ncore=$<TARGET_FILE:softloaf_core>\nbrowser_cache=${SOFTLOAF_WEBVIEW2_CACHE}\n")
endif()
if(APPLE AND SOFTLOAF_DESKTOP)
    enable_language(OBJCXX)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(SoftLoafSpectral MACOSX_BUNDLE desktop/macos/main.mm)
    foreach(language IN ITEMS en zh-Hans)
        set(softloaf_localized_info "${CMAKE_SOURCE_DIR}/desktop/macos/${language}.lproj/InfoPlist.strings")
        set_source_files_properties("${softloaf_localized_info}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${language}.lproj")
        target_sources(SoftLoafSpectral PRIVATE "${softloaf_localized_info}")
    endforeach()
    target_compile_features(SoftLoafSpectral PRIVATE cxx_std_23)
    target_compile_options(SoftLoafSpectral PRIVATE -fobjc-arc -Wall -Wextra)
    target_link_libraries(SoftLoafSpectral PRIVATE "-framework Cocoa" "-framework WebKit")
    # User-authorized local branding is optional, like the material assets.
    set(SOFTLOAF_APP_ICON "${CMAKE_SOURCE_DIR}/assets/branding/SoftLoafSpectral.icns")
    if(EXISTS "${SOFTLOAF_APP_ICON}")
        set_source_files_properties("${SOFTLOAF_APP_ICON}" PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
        target_sources(SoftLoafSpectral PRIVATE "${SOFTLOAF_APP_ICON}")
        set_target_properties(SoftLoafSpectral PROPERTIES MACOSX_BUNDLE_ICON_FILE "SoftLoafSpectral.icns")
    endif()
    set_target_properties(SoftLoafSpectral PROPERTIES
        OUTPUT_NAME "SoftLoaf Spectral"
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/desktop/macos/Info.plist.in")
    add_dependencies(SoftLoafSpectral softloaf_core softloaf_metal softloaf_imageio)
    if(TARGET softloaf_rawtherapee_cli)
        add_dependencies(SoftLoafSpectral softloaf_rawtherapee_cli)
    endif()
    if(BUILD_TESTING)
        add_executable(softloaf_content_layout_test desktop/macos/content_layout_test.mm)
        target_compile_features(softloaf_content_layout_test PRIVATE cxx_std_23)
        target_compile_options(softloaf_content_layout_test PRIVATE -fobjc-arc)
        target_link_libraries(softloaf_content_layout_test PRIVATE "-framework Cocoa")
        add_test(NAME desktop_content_layout COMMAND softloaf_content_layout_test)
    endif()
    # A local development bundle, intentionally tied to the configured checkout
    # and Python environment. External material assets are never copied into it.
    configure_file(desktop/macos/runtime.plist.in runtime.plist @ONLY)
    # POST_BUILD alone does not run when only the independently loaded core
    # changes. Refresh resources on every default build, after both targets.
    add_custom_target(softloaf_desktop_runtime ALL
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:SoftLoafSpectral>/Contents/Resources"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_BINARY_DIR}/runtime.plist"
            "$<TARGET_BUNDLE_DIR:SoftLoafSpectral>/Contents/Resources/runtime.plist"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:softloaf_core>"
            "$<TARGET_BUNDLE_DIR:SoftLoafSpectral>/Contents/Resources/$<TARGET_FILE_NAME:softloaf_core>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:softloaf_metal>"
            "$<TARGET_BUNDLE_DIR:SoftLoafSpectral>/Contents/Resources/$<TARGET_FILE_NAME:softloaf_metal>"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:softloaf_imageio>"
            "$<TARGET_BUNDLE_DIR:SoftLoafSpectral>/Contents/Resources/$<TARGET_FILE_NAME:softloaf_imageio>"
        DEPENDS SoftLoafSpectral softloaf_core softloaf_metal softloaf_imageio)
endif()

# Honor this checkout's private external-data preference for test temporaries.
if(BUILD_TESTING AND EXISTS "${CMAKE_SOURCE_DIR}/local/storage-settings.json")
    file(READ "${CMAKE_SOURCE_DIR}/local/storage-settings.json" SOFTLOAF_STORAGE_SETTINGS)
    string(JSON SOFTLOAF_TEST_DATA_ROOT GET "${SOFTLOAF_STORAGE_SETTINGS}" data_root)
    set_tests_properties(core_hand_calculated PROPERTIES ENVIRONMENT "TMPDIR=${SOFTLOAF_TEST_DATA_ROOT}/tmp")
    if(TEST desktop_content_layout)
        set_tests_properties(desktop_content_layout PROPERTIES ENVIRONMENT "TMPDIR=${SOFTLOAF_TEST_DATA_ROOT}/tmp")
    endif()
endif()
